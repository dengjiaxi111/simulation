cmake_minimum_required(VERSION 3.16)
project(pid_observer)
set(CMAKE_CXX_STANDARD 17)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2-register REQUIRED)
find_package(gz-transport13 REQUIRED)
add_library(pid_observer SHARED observer.cpp)
target_link_libraries(pid_observer gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(telemetry_bridge bridge.cpp)
ament_target_dependencies(telemetry_bridge rclcpp std_msgs)
target_link_libraries(telemetry_bridge gz-transport13::gz-transport13)
